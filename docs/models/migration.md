# Migration

## Example Usage

```typescript
import { Migration } from "@paygentic/sdk/models";

let value: Migration = {
  fromVersionNumber: 524178,
  toVersionNumber: 97949,
  effectiveAt: new Date("2024-03-01T00:25:32.185Z"),
  entitlementsEffectiveAt: new Date("2026-09-16T08:43:55.304Z"),
  changed: true,
};
```
