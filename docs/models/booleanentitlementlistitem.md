# BooleanEntitlementListItem

Common fields shared by all entitlement list items. List items use `entitlementId` (not `id`) to preserve the original public field name on `/v1/entitlements`. The get-by-id endpoint returns the same object with a top-level `id` and `object: "entitlement"` instead.

## Example Usage

```typescript
import { BooleanEntitlementListItem } from "@paygentic/sdk/models";

let value: BooleanEntitlementListItem = {
  entitlementId: "<id>",
  customerId: "<id>",
  featureId: "<id>",
  featureKey: "<value>",
  featureType: "boolean",
  productId: "<id>",
  subscriptionId: "<id>",
  status: "canceled",
  activeFrom: new Date("2026-04-27T17:56:46.611Z"),
  activeTo: null,
  hasAccess: false,
  metadata: {
    "key": "<value>",
  },
  config: {
    "key": "<value>",
  },
};
```
